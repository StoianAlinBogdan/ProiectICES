PREFIX party: <http://www.party.com/ontologies/party.owl#>
SELECT ?p ?view
	WHERE { ?p a party:Party .
	?p party:hasDunBradstreetView ?view .
	?view a party:DunBradstreetView .
}

SELECT (str(?partyId) as ?id) ( str(?partyName) as ?nume)
	WHERE { ?p a party:Party;
		party:hasPartyId ?partyId;
		party:hasPartyName ?partyName .
}


SELECT DISTINCT ?p (str(?partyName) as ?nume) (str(?dunsAddress) as ?adresa) ?submission ?layer
WHERE {
	?p a party:Party;
	party:hasPartyName ?partyName;
	party:hasDunBradstreetView ?dunsView.
	?dunsView party:hasDunsAddress ?dunsAddress .
